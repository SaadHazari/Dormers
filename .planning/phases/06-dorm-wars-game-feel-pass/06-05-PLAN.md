type NinesliceProps = {

 * Implementation: 4 corner SVG snippets + horizontal/vertical edge tiles via CSS repeating-linear-gradient

 */
export function NinesliceStampedBorder({ children, color = 'currentColor', borderWidth: _borderWidth = 2, className }: NinesliceProps) {
  return (
    <span className={className} style={{ position: 'relative', display: 'inline-block', padding: 6, color }}>
      {/* Top-left corner */}

      {/* Top-right corner */}
