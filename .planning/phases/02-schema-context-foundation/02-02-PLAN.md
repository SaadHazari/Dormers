  role: OpsRole
  label: string        // "Main Kitchen", "Rider 1"
  is_active: boolean

  revoked_at: string | null
}

/** A token is valid when it exists, is_active is true, and revoked_at is null. */
export function isTokenValid(t: OpsToken | null): t is OpsToken {
  return t !== null && t.is_active && t.revoked_at === null
}

```typescript
// Ops context — token validation use-case.
// Queries ops_tokens table via service-role client.

import { createAdminSupabaseClient } from '@/infra/supabase/admin-client'
import type { OpsToken, OpsRole } from '@/contexts/ops/domain/ops-token'
import { isTokenValid } from '@/contexts/ops/domain/ops-token'

/**
 * Look up an ops token by its secret string. Returns the token row if valid
 * (active + not revoked), or null if not found / inactive / revoked.
 *
 * Optionally filters by role — e.g. kitchen pages pass role='kitchen' to
 * reject rider tokens even if valid.
 */
export async function validateOpsToken(
  tokenString: string,
  requiredRole?: OpsRole,
): Promise<OpsToken | null> {
  const supabase = createAdminSupabaseClient()

  const { data, error } = await supabase
    .from('ops_tokens')
    .select('id, token, role, label, is_active, created_at, revoked_at')
    .eq('token', tokenString)
    .single()

  if (error || !data) return null

  const opsToken = data as OpsToken

  if (!isTokenValid(opsToken)) return null
  if (requiredRole && opsToken.role !== requiredRole) return null

  return opsToken
}
```
