- STEP = `CARD_WIDTH_MOBILE + CARD_GAP` (= 136)
- Drag constraints: compute in `useEffect` after mount using `containerRef.current.offsetWidth`:
  ```typescript
  const [constraints, setConstraints] = useState({ left: 0, right: 0 });
  useEffect(() => {
    if (!containerRef.current) return;
    const containerW = containerRef.current.offsetWidth;
    const trackW = STEP * slots.length;
    setConstraints({ left: -(trackW - containerW + PEEK), right: PEEK });
  }, [slots.length]);
  ```

Snap-to-card (per Research — use modifyTarget, NOT manual animate in onDragEnd):
```typescript
dragTransition={{
  bounceStiffness: 300,
  bounceDamping: 30,
  modifyTarget: (target: number) => Math.round(target / STEP) * STEP,
}}
```

onDragEnd state sync (NO animate call — modifyTarget handles position):
```typescript
function handleDragEnd() {
  const snappedX = x.get();
  const idx = Math.max(0, Math.min(Math.round(-snappedX / STEP), slots.length - 1));
  const dish = slots[idx];
  if (dish) setSelectedDay(dish.dayOfWeek);
}
```

Auto-scroll to today on mount (GALL-04, per D-11):
```typescript
useEffect(() => {
  if (selectedDay === null) return;
  const idx = slots.findIndex(d => d && d.dayOfWeek === selectedDay);
  if (idx < 0) return;
  const targetX = -(idx * STEP);
  animate(x, targetX, { type: "spring", stiffness: 300, damping: 30 });
}, []); // mount only
```

Also: when a card is tapped (not dragged), animate the gallery to center that card:
```typescript
function handleCardSelect(dayOfWeek: number) {
  setSelectedDay(dayOfWeek);
  const idx = slots.findIndex(d => d && d.dayOfWeek === dayOfWeek);
  if (idx < 0) return;
  const targetX = -(idx * STEP);
  const clampedX = Math.max(constraints.left, Math.min(targetX, constraints.right));
  animate(x, clampedX, { type: "spring", stiffness: 300, damping: 30 });
}
```

Gallery JSX structure:
```tsx
<div ref={containerRef} className="overflow-hidden relative" style={{ paddingLeft: PEEK, paddingRight: PEEK }}>
  <motion.div
    ref={scope}
    drag="x"
    style={{ x }}
    dragConstraints={constraints}
    dragElastic={0.1}
    dragDirectionLock
    dragTransition={{
      bounceStiffness: 300,
      bounceDamping: 30,
      modifyTarget: (target: number) => Math.round(target / STEP) * STEP,
    }}
    onDragEnd={handleDragEnd}
    className="flex"
    style={{ x, gap: CARD_GAP }}
  >
    {slots.map((dish, i) =>
      dish ? (
        <DishCard
          key={dish.id}
          dish={dish}
          isSelected={dish.dayOfWeek === selectedDay}
          onSelect={() => handleCardSelect(dish.dayOfWeek)}
        />
      ) : (
        <EmptyCard key={`empty-${i}`} dayIndex={i} />
      )
    )}
  </motion.div>
</div>
```
