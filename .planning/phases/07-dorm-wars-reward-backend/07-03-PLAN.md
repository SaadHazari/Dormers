```typescript
import { randomInt } from 'crypto'

export function mysteryDropValue(): number {
  const roll = randomInt(0, 100)   // 0..99
  if (roll < 50) return randomInt(30, 71)     // 30..70   (50%)
  if (roll < 85) return randomInt(71, 121)    // 71..120  (35%)
  return randomInt(121, 151)                  // 121..150 (15%)
}

export function dailyDropValue(): number {
  const roll = randomInt(0, 100)
  if (roll < 60) return randomInt(1, 11)      // 1..10    (60%)
  if (roll < 90) return randomInt(11, 51)     // 11..50   (30%)
  return randomInt(51, 201)                   // 51..200  (10%)
}
```

**Note on `crypto.randomInt`:** Node 14.10+, supported on Netlify (Node 18+). Do NOT replace with `Math.random()` — Pitfall #6 + standard-stack table both call this out.

```typescript
/**
 * Count of inviter's referrals that converted to paid since the given
 * subscription's start_date. Shared source-of-truth between the Layer 2
 * awarder (src/lib/dorm-wars/awarder.ts) and the dorm-wars hub
 * (src/app/dashboard/dorm-wars/page.tsx) per RESEARCH Decision #10.
 */
export async function getCycleRecruits(
  sb: ReturnType<typeof import('@supabase/supabase-js').createClient>,
  customerId: string,
  subscriptionId: string,
): Promise<number> {
  const { data: sub } = await sb
    .from('subscriptions')
    .select('start_date')
    .eq('id', subscriptionId)
    .maybeSingle()
  if (!sub) return 0

  const { count } = await sb
    .from('referrals')
    .select('id', { count: 'exact', head: true })
    .eq('inviter_user_id', customerId)
    .eq('status', 'converted')
    .gte('converted_at', sub.start_date)
  return count ?? 0
}
```

```typescript
// src/lib/dorm-wars/awarder.ts
// Layer 2 + Layer 3 reward awarder. Called inline from
// creditInviterOnConversion (src/app/r/[cid]/actions.ts) AND from the
// monthly-cap early-return — Layer 3 tiers fire on lifetime count regardless
// of the Layer 1 monthly cap. (RESEARCH Decision #2.)
//
// Idempotency: cycle_rewards(customer_id, subscription_id, milestone) UNIQUE
// and lifetime_rewards(customer_id, tier) UNIQUE mean re-fires are safe.

import { createClient } from '@supabase/supabase-js'
import { mysteryDropValue } from './rng'
import { CYCLE_MILESTONES } from './constants'
import { getCycleRecruits } from '@/utils/supabase/queries'

type AdminClient = ReturnType<typeof createClient>

function admin(): AdminClient {
  return createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!,
  )
}

async function depositCredit(
  sb: AdminClient,
  customerId: string,
  amountAed: number,
  source: string,
  // cycleRewardId is logged only — no FK on credits → cycle_rewards in Phase 7
  _cycleRewardId: string,
): Promise<void> {
  await sb.from('credits').insert({
    customer_id: customerId,
    amount_aed: amountAed,
    source,
    status: 'approved',
  })
}

export async function awardCycleAndTierRewards(
  customerId: string,
  subscriptionId: string | null,
): Promise<void> {
  const sb = admin()

  // ── Layer 2: cycle milestones ──
  // Per Decision #9: skip Layer 2 entirely when no active sub.
  // Per Pitfall #7: Layer 2 ticks regardless of Layer 1 monthly cap — INTENTIONAL.
  if (subscriptionId) {
    const cycleRecruits = await getCycleRecruits(sb, customerId, subscriptionId)

    for (const m of CYCLE_MILESTONES) {
      if (cycleRecruits < m.at) break // ascending — short-circuit
      const value = m.kind === 'mystery_drop' ? mysteryDropValue() : m.value

      const { data: inserted } = await sb
        .from('cycle_rewards')
        .insert({
          customer_id: customerId,
          subscription_id: subscriptionId,
          milestone: m.at,
          kind: m.kind,
          value_aed: value,
        })
        .select('id')
        .maybeSingle() // null on UNIQUE conflict

      if (!inserted) continue // already awarded — idempotent

      // Side effects on first-award only:
      if (m.kind === 'mystery_drop' || m.kind === 'free_week' || m.kind === 'free_month') {
        await depositCredit(sb, customerId, value!, `cycle_milestone_${m.at}`, inserted.id)
      }
      if (m.kind === 'cash_and_skips') {
        await depositCredit(sb, customerId, 500, 'cycle_milestone_15', inserted.id)
        await sb.rpc('increment_bonus_skips', { p_sub_id: subscriptionId, p_amount: 5 })
      }
      if (m.kind === 'dorm_weekend') {
        // Decision #8: stub — row written, no credit, ops-visible log.
        console.log(`🏆 DORM WEEKEND unlocked for customer ${customerId} — manual fulfilment needed`)
      }
    }
  }

  // LAYER 3 TIER LOOP APPENDED IN 07-04
}
```

**Key implementation guardrails:**
- `value!` non-null assertion on line that calls `depositCredit` — valid because all three kinds (`mystery_drop`, `free_week`, `free_month`) have non-null `value` at that point (mystery resolved by RNG, free_week/month from constants).
- `.maybeSingle()` after `.insert(...).select('id')` returns `null` on UNIQUE conflict — that's the idempotency mechanism. Do NOT use `.single()` (throws on conflict).
- `console.log` on dorm_weekend is the OPS visibility hook per Decision #8. Keep it.
- Do NOT add a try/catch around the whole loop — let errors propagate so the webhook gets the 5xx and Stripe retries.
